theory Coma
  
  (* use why3.BuiltIn.BuiltIn *)

  predicate gt_log_Int (self:int) (other:int) = other < self
  
  (* meta rewrite_def predicate gt_log_Int *)
  
  predicate unchanged_byte (byte:int) =
    byte = 42 \/
    byte = 45 \/
    byte = 46 \/
    48 <= byte /\ byte <= 57 \/
    65 <= byte /\ byte <= 90 \/ byte = 95 \/ 97 <= byte /\ byte <= 122
  
  axiom unchanged_byte_spec :
    forall byte:int [unchanged_byte byte].
     unchanged_byte byte <->
     byte = 42 \/
     byte = 45 \/
     byte = 46 \/
     48 <= byte /\ byte <= 57 \/
     65 <= byte /\ byte <= 90 \/ byte = 95 \/ 97 <= byte /\ byte <= 122
  
  function safe_run_end (seq t) int : int
  
  axiom safe_run_end_def :
    forall bytes:seq t, start:int [safe_run_end bytes start].
     0 <= start /\ start <= length bytes ->
     safe_run_end bytes start =
     (if start < length bytes /\ unchanged_byte (t'int (get bytes start))
      then safe_run_end bytes (start + 1) else start)
  
  axiom safe_run_end_spec :
    forall bytes:seq t, start:int [safe_run_end bytes start].
     0 <= start /\ start <= length bytes ->
     start <= safe_run_end bytes start /\
     safe_run_end bytes start <= length bytes
  
  axiom safe_run_end_spec'0 :
    forall bytes:seq t, start:int [safe_run_end bytes start].
     0 <= start /\ start <= length bytes ->
     (forall i:int.
       start <= i /\ i < safe_run_end bytes start ->
       unchanged_byte (t'int (get bytes i)))
  
  axiom safe_run_end_spec'1 :
    forall bytes:seq t, start:int [safe_run_end bytes start].
     0 <= start /\ start <= length bytes ->
     safe_run_end bytes start = length bytes \/
     not unchanged_byte (t'int (get bytes (safe_run_end bytes start)))
  
  axiom safe_run_end_spec'2 :
    forall bytes:seq t, start:int [safe_run_end bytes start].
     0 <= start /\ start <= length bytes ->
     start < length bytes /\ unchanged_byte (t'int (get bytes start)) ->
     start < safe_run_end bytes start
  
  function index_Mapping_u8_Int (self:t -> int) (a:t) : int = get1 self a
  
  (* meta rewrite_def function index_Mapping_u8_Int *)
  
  function map_u8 (seq t) (t -> int) : seq int
  
  axiom map_u8_spec :
    forall self:seq t, m:t -> int [map_u8 self m].
     length (map_u8 self m) = length self
  
  axiom map_u8_spec'0 :
    forall self:seq t, m:t -> int [map_u8 self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get (map_u8 self m) i = index_Mapping_u8_Int m (get self i)
  
  function uppercase_hex (digit:int) : int =
    if digit < 10 then 48 + digit else 55 + digit
  
  axiom uppercase_hex_spec :
    forall digit:int [uppercase_hex digit].
     uppercase_hex digit = (if digit < 10 then 48 + digit else 55 + digit)
  
  function escaped_byte_chunk (byte:int) : seq int =
    if byte = 32 then singleton 43
    else (singleton 37 ++ singleton (uppercase_hex (div byte 16)))
         ++ singleton (uppercase_hex (mod byte 16))
  
  function encoded_chunks (seq t) : seq (seq int)
  
  axiom encoded_chunks_def :
    forall bytes:seq t [encoded_chunks bytes].
     encoded_chunks bytes =
     (if length bytes = 0 then empty: seq (seq int)
      else let byte = t'int (get bytes 0) in
           if unchanged_byte byte
           then let end' = safe_run_end bytes 0 in
                singleton (map_u8 bytes[0..end'] (fun (b:t) -> t'int b))
                ++ encoded_chunks bytes[end'..length bytes]
           else singleton (escaped_byte_chunk byte)
                ++ encoded_chunks bytes[1..length bytes])
  
  axiom encoded_chunks_spec :
    forall bytes:seq t [encoded_chunks bytes].
     length bytes = 0 <-> length (encoded_chunks bytes) = 0
  
  axiom encoded_chunks_spec'0 :
    forall bytes:seq t [encoded_chunks bytes].
     gt_log_Int (length bytes) 0 ->
     1 <= length (encoded_chunks bytes) /\
     length (encoded_chunks bytes) <= length bytes
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  constant bytes : seq t
  
  constant byte : int
  
  predicate encoded_chunks_first_escaped (seq t) int
  
  goal vc_encoded_chunks_first_escaped :
    gt_log_Int (length bytes) 0 /\
    not unchanged_byte (t'int (get bytes 0)) /\ byte = t'int (get bytes 0) ->
    true /\
    encoded_chunks bytes =
    (singleton (escaped_byte_chunk byte)
     ++ encoded_chunks bytes[1..length bytes])
  
end
