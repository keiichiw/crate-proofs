  goal vc_slice_after_equals :
    forall input:slice t, eq:t1, end':t1.
     t'int eq < t'int end' /\ t'int end' <= length (view input) ->
     in_bounds (t'int eq + t'int (1:t1)) /\
     (forall result:t1.
       t'int result = (t'int eq + t'int (1:t1)) ->
       (t'int result <= t'int end' /\ t'int end' <= length (view input)) /\
       (forall result1:slice t.
         view result1 = (view input)[t'int result..t'int end'] ->
         view result1 = (view input)[(t'int eq + 1)..t'int end']))
  
end
