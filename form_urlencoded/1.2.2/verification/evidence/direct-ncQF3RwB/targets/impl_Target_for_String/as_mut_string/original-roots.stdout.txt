  (* use why3.BuiltIn.BuiltIn *)

  (* use creusot.prelude.Any *)
  
  type t_String
  
  predicate inv_String t_String
  
  predicate invariant_refmut_String (self:t t_String) =
    inv_String (self.current) /\ inv_String (self.final)
  
  (* meta rewrite_def predicate invariant_refmut_String *)
  
  predicate inv_refmut_String (_1:t t_String) = invariant_refmut_String _1
  
  (* meta rewrite_def predicate inv_refmut_String *)
  
  predicate resolve_refmut_String (_1:t t_String) = _1.final = _1.current
  
  (* meta rewrite_def predicate resolve_refmut_String *)
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  goal vc_as_mut_string_String :
    forall self:t t_String.
     inv_refmut_String self ->
     (forall fin:t_String.
       inv_String fin ->
       (forall fin1:t_String.
         inv_String fin1 ->
         (resolve_refmut_String (t'mk fin1 fin (get_id self)) ->
          (resolve_refmut_String (t'mk fin (self.final) (self.id)) ->
           inv_refmut_String
           (t'mk (self.current) fin1
            (get_id (t'mk (self.current) fin (get_id self)))) /\
           fin1 = self.final) /\
          inv_refmut_String (t'mk fin (self.final) (self.id))) /\
         inv_refmut_String (t'mk fin1 fin (get_id self))) /\
       inv_String (self.current)) /\
     inv_String (self.current)
  
end
