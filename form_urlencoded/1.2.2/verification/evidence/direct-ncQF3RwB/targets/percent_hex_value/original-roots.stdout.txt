theory Coma
  
  (* use why3.BuiltIn.BuiltIn *)
  
  (* use why3.Bool.Bool *)
  
  (* use why3.Unit.Unit *)
  
  (* use creusot.int.UInt8 *)
  
  (* use creusot.prelude.Any *)
  
  type t_Option_u8 =
    | None
    | Some t
  
  function percent_hex_value_model (byte:int) : int =
    if 48 <= byte /\ byte <= 57 then byte - 48
    else if 65 <= byte /\ byte <= 70 then byte - 55
         else if 97 <= byte /\ byte <= 102 then byte - 87 else (- 1)
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  goal vc_percent_hex_value :
    forall byte:t.
     forall _6:bool, _7:bool, _8:bool, _9:bool.
      (forall _81:bool, _91:bool.
        (le (48:t) byte ->
         le byte (57:t) ->
         in_bounds (t'int byte - t'int (48:t)) /\
         (forall result:t.
           t'int result = (t'int byte - t'int (48:t)) ->
           percent_hex_value_model (t'int byte) = t'int result /\
           0 <= t'int result /\ t'int result <= 15)) /\
        (not le (48:t) byte /\
         (if le (48:t) byte then _81 = True else _81 = False) /\
         _91 = (any_l (): bool) \/
         le (48:t) byte /\
         not le byte (57:t) /\
         (if le (48:t) byte then _81 = True else _81 = False) /\
         (if le byte (57:t) then _91 = True else _91 = False) ->
         le (65:t) byte ->
         le byte (70:t) ->
         in_bounds (t'int byte - t'int (65:t)) /\
         (forall result:t.
           t'int result = (t'int byte - t'int (65:t)) ->
           in_bounds (t'int result + t'int (10:t)) /\
           (forall result1:t.
             t'int result1 = (t'int result + t'int (10:t)) ->
             percent_hex_value_model (t'int byte) = t'int result1 /\
             0 <= t'int result1 /\ t'int result1 <= 15)))) /\
      ((exists _81:bool, _91:bool.
         (not le (48:t) byte /\
          (if le (48:t) byte then _81 = True else _81 = False) /\
          _91 = (any_l (): bool) \/
          le (48:t) byte /\
          not le byte (57:t) /\
          (if le (48:t) byte then _81 = True else _81 = False) /\
          (if le byte (57:t) then _91 = True else _91 = False)) /\
         (not le (65:t) byte /\
          (if le (65:t) byte then _6 = True else _6 = False) /\
          _7 = (any_l (): bool) /\ _8 = _81 /\ _9 = _91 \/
          le (65:t) byte /\
          not le byte (70:t) /\
          (if le (65:t) byte then _6 = True else _6 = False) /\
          (if le byte (70:t) then _7 = True else _7 = False) /\
          _8 = _81 /\ _9 = _91)) ->
       (not le (97:t) byte -> percent_hex_value_model (t'int byte) = (- 1)) /\
       (le (97:t) byte ->
        (not le byte (102:t) -> percent_hex_value_model (t'int byte) = (- 1)) /\
        (le byte (102:t) ->
         in_bounds (t'int byte - t'int (97:t)) /\
         (forall result:t.
           t'int result = (t'int byte - t'int (97:t)) ->
           in_bounds (t'int result + t'int (10:t)) /\
           (forall result1:t.
             t'int result1 = (t'int result + t'int (10:t)) ->
             percent_hex_value_model (t'int byte) = t'int result1 /\
             0 <= t'int result1 /\ t'int result1 <= 15)))))
  
end
