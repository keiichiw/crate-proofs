  predicate gt_log_Int (self:int) (other:int) = other < self
  
  (* meta rewrite_def predicate gt_log_Int *)
  
  predicate ge_log_Int (self:int) (other:int) = other <= self
  
  (* meta rewrite_def predicate ge_log_Int *)
  
  predicate should_encode_model (mask:seq t) (byte:t1) =
    ge_log_Int (t'int byte) 128 \/
    not bw_and (get mask (div (t'int byte) 32))
        (shl (1:t) (of_int (t'int (bw_and1 byte (31:t1))))) = (0:t)
  
  function uppercase_hex (digit:int) : int =
    if digit < 10 then 48 + digit else 55 + digit
  
  axiom uppercase_hex_spec :
    forall digit:int [uppercase_hex digit].
     uppercase_hex digit = (if digit < 10 then 48 + digit else 55 + digit)
  
  function uppercase_hex'0 (digit:int) : int = uppercase_hex digit
  
  function encoded_byte_model (byte:int) : seq int =
    (singleton 37 ++ singleton (uppercase_hex'0 (div byte 16)))
    ++ singleton (uppercase_hex'0 (mod byte 16))
  
  function safe_run_end (seq t1) int (seq t) : int
  
  axiom safe_run_end_spec :
    forall input:seq t1, start:int, mask:seq t
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start <= safe_run_end input start mask /\
     safe_run_end input start mask <= length input
  
  axiom safe_run_end_spec'0 :
    forall input:seq t1, start:int, mask:seq t
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     (forall i:int.
       start <= i /\ i < safe_run_end input start mask ->
       not should_encode_model mask (get input i))
  
  axiom safe_run_end_spec'1 :
    forall input:seq t1, start:int, mask:seq t
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     safe_run_end input start mask = length input \/
     should_encode_model mask (get input (safe_run_end input start mask))
  
  axiom safe_run_end_spec'2 :
    forall input:seq t1, start:int, mask:seq t
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start < length input /\ not should_encode_model mask (get input start) ->
     start < safe_run_end input start mask
  
  function index_Mapping_u8_Int (self:t1 -> int) (a:t1) : int = get1 self a
  
  (* meta rewrite_def function index_Mapping_u8_Int *)
  
  function map_u8 (seq t1) (t1 -> int) : seq int
  
  axiom map_u8_spec :
    forall self:seq t1, m:t1 -> int [map_u8 self m].
     length (map_u8 self m) = length self
  
  axiom map_u8_spec'0 :
    forall self:seq t1, m:t1 -> int [map_u8 self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get (map_u8 self m) i = index_Mapping_u8_Int m (get self i)
  
  function byte_sequence_as_ints (bytes:seq t1) : seq int =
    map_u8 bytes (fun (byte:t1) -> t'int byte)
  
  function encoded_chunks (seq t) (seq t1) : seq (seq int)
  
  axiom encoded_chunks_def :
    forall mask:seq t, input:seq t1 [encoded_chunks mask input].
     length mask = 4 ->
     encoded_chunks mask input =
     (if length input = 0 then empty: seq (seq int)
      else if should_encode_model mask (get input 0)
           then singleton (encoded_byte_model (t'int (get input 0)))
                ++ encoded_chunks mask input[1..length input]
           else let end' = safe_run_end input 0 mask in
                singleton (byte_sequence_as_ints input[0..end'])
                ++ encoded_chunks mask input[end'..length input])
  
  axiom encoded_chunks_spec :
    forall mask:seq t, input:seq t1 [encoded_chunks mask input].
     length mask = 4 ->
     (if length input = 0 then length (encoded_chunks mask input) = 0
      else gt_log_Int (length (encoded_chunks mask input)) 0 /\
           (if should_encode_model mask (get input 0)
            then get (encoded_chunks mask input) 0 =
                 encoded_byte_model (t'int (get input 0)) /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[1..length input]
            else let end' = safe_run_end input 0 mask in
                 get (encoded_chunks mask input) 0 =
                 byte_sequence_as_ints input[0..end'] /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[end'..length input]))
  
  predicate encoded_chunks_first_escaped (seq t) (seq t1)
  
  axiom encoded_chunks_first_escaped_spec :
    forall mask:seq t, input:seq t1
     [encoded_chunks_first_escaped mask input].
     gt_log_Int (length input) 0 /\ length mask = 4 ->
     should_encode_model mask (get input 0) ->
     encoded_chunks_first_escaped mask input
  
  axiom encoded_chunks_first_escaped_spec'0 :
    forall mask:seq t, input:seq t1.
     gt_log_Int (length input) 0 /\ length mask = 4 ->
     should_encode_model mask (get input 0) ->
     encoded_chunks mask input =
     (singleton (encoded_byte_model (t'int (get input 0)))
      ++ encoded_chunks mask input[1..length input])
  
  predicate well_founded_relation_Int (self:int) (other:int) =
    ge_log_Int self 0 /\ gt_log_Int self other
  
  (* meta rewrite_def predicate well_founded_relation_Int *)
  
  predicate encoded_chunks_first_run (seq t) (seq t1) int
  
  axiom encoded_chunks_first_run_spec :
    forall mask:seq t, input:seq t1, end':int
     [encoded_chunks_first_run mask input end'].
     gt_log_Int (length input) 0 /\ length mask = 4 ->
     not should_encode_model mask (get input 0) ->
     end' = safe_run_end input 0 mask ->
     encoded_chunks_first_run mask input end'
  
  axiom encoded_chunks_first_run_spec'0 :
    forall mask:seq t, input:seq t1, end':int.
     gt_log_Int (length input) 0 /\ length mask = 4 ->
     not should_encode_model mask (get input 0) ->
     end' = safe_run_end input 0 mask ->
     encoded_chunks mask input =
     (singleton (byte_sequence_as_ints input[0..end'])
      ++ encoded_chunks mask input[end'..length input])

  (* meta select_lsinst "all" *)
  
  constant mask : seq t
  
  constant input : seq t1
  
  predicate encoded_chunk_count_bounds (seq t) (seq t1)
  
  goal vc_encoded_chunk_count_bounds :
    length mask = 4 ->
    (if length input = 0
     then true /\
          length (encoded_chunks mask input) <= length input /\
          (length input = 0 <-> length (encoded_chunks mask input) = 0) /\
          (gt_log_Int (length input) 0 ->
           ge_log_Int (length (encoded_chunks mask input)) 1)
     else if should_encode_model mask (get input 0)
          then let tail = input[1..length input] in
               ((gt_log_Int (length input) 0 /\ length mask = 4) /\
                should_encode_model mask (get input 0)) /\
               (encoded_chunks_first_escaped mask input /\
                encoded_chunks mask input =
                (singleton (encoded_byte_model (t'int (get input 0)))
                 ++ encoded_chunks mask input[1..length input]) ->
                (if encoded_chunks_first_escaped mask input
                 then (length mask = 4 /\
                       well_founded_relation_Int (length input) (length tail)) /\
                      (encoded_chunk_count_bounds mask tail /\
                       length (encoded_chunks mask tail) <= length tail /\
                       (length tail = 0 <->
                        length (encoded_chunks mask tail) = 0) /\
                       (gt_log_Int (length tail) 0 ->
                        ge_log_Int (length (encoded_chunks mask tail)) 1) ->
                       encoded_chunk_count_bounds mask tail /\
                       length (encoded_chunks mask input) <= length input /\
                       (length input = 0 <->
                        length (encoded_chunks mask input) = 0) /\
                       (gt_log_Int (length input) 0 ->
                        ge_log_Int (length (encoded_chunks mask input)) 1))
                 else false /\
                      length (encoded_chunks mask input) <= length input /\
                      (length input = 0 <->
                       length (encoded_chunks mask input) = 0) /\
                      (gt_log_Int (length input) 0 ->
                       ge_log_Int (length (encoded_chunks mask input)) 1)))
          else ((0 <= 0 /\ 0 <= length input) /\ length mask = 4) /\
               ((0 <= safe_run_end input 0 mask /\
                 safe_run_end input 0 mask <= length input) /\
                (forall i:int.
                  0 <= i /\ i < safe_run_end input 0 mask ->
                  not should_encode_model mask (get input i)) /\
                (safe_run_end input 0 mask = length input \/
                 should_encode_model mask
                 (get input (safe_run_end input 0 mask))) /\
                (0 < length input /\
                 not should_encode_model mask (get input 0) ->
                 0 < safe_run_end input 0 mask) ->
                (let end' = safe_run_end input 0 mask in
                 let tail = input[end'..length input] in
                 ((gt_log_Int (length input) 0 /\ length mask = 4) /\
                  not should_encode_model mask (get input 0) /\
                  end' = safe_run_end input 0 mask) /\
                 (encoded_chunks_first_run mask input end' /\
                  encoded_chunks mask input =
                  (singleton (byte_sequence_as_ints input[0..end'])
                   ++ encoded_chunks mask input[end'..length input]) ->
                  (if encoded_chunks_first_run mask input end'
                   then (length mask = 4 /\
                         well_founded_relation_Int (length input)
                         (length tail)) /\
                        (encoded_chunk_count_bounds mask tail /\
                         length (encoded_chunks mask tail) <= length tail /\
                         (length tail = 0 <->
                          length (encoded_chunks mask tail) = 0) /\
                         (gt_log_Int (length tail) 0 ->
                          ge_log_Int (length (encoded_chunks mask tail)) 1) ->
                         encoded_chunk_count_bounds mask tail /\
                         length (encoded_chunks mask input) <= length input /\
                         (length input = 0 <->
                          length (encoded_chunks mask input) = 0) /\
                         (gt_log_Int (length input) 0 ->
                          ge_log_Int (length (encoded_chunks mask input)) 1))
                   else false /\
                        length (encoded_chunks mask input) <= length input /\
                        (length input = 0 <->
                         length (encoded_chunks mask input) = 0) /\
                        (gt_log_Int (length input) 0 ->
                         ge_log_Int (length (encoded_chunks mask input)) 1))))))
