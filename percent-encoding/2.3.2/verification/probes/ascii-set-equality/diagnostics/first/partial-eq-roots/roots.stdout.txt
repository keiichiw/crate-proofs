theory Coma
  
  (* use why3.BuiltIn.BuiltIn *)
  
  (* use why3.Bool.Bool *)
  
  (* use why3.Unit.Unit *)
  
  (* use creusot.int.UInt64 *)
  
  (* use creusot.int.UInt32 *)
  
  (* use creusot.slice.Slice64 *)
  
  (* use creusot.prelude.Any *)
  
  type t_AsciiSet =
    | T_AsciiSet'mk (mask:slice t)
  
  (* meta model_record function mask *)
  
  predicate inv_Seq_u32 (_1:seq t) = true
  
  (* meta rewrite_def predicate inv_Seq_u32 *)
  
  predicate invariant_array_u32_4 (self:slice t) =
    inv_Seq_u32 (view self) /\ length (view self) = t'int (4:t1)
  
  predicate inv_array_u32_4 (_1:slice t) = invariant_array_u32_4 _1
  
  (* meta rewrite_def predicate inv_array_u32_4 *)
  
  predicate inv_AsciiSet t_AsciiSet
  
  axiom inv_axiom :
    forall x:t_AsciiSet [inv_AsciiSet x].
     inv_AsciiSet x <-> inv_array_u32_4 (x.mask)
  
  predicate invariant_ref_AsciiSet (self:t_AsciiSet) = inv_AsciiSet self
  
  (* meta rewrite_def predicate invariant_ref_AsciiSet *)
  
  predicate inv_ref_AsciiSet (_1:t_AsciiSet) = invariant_ref_AsciiSet _1
  
  (* meta rewrite_def predicate inv_ref_AsciiSet *)
  
  function mask_words (self:t_AsciiSet) : seq t =
    create 4 (fun (i:int) -> get (view (self.mask)) i)
  
  axiom mask_words_spec :
    forall self:t_AsciiSet [mask_words self]. length (mask_words self) = 4
  
  function deep_model_AsciiSet (self:t_AsciiSet) : seq t = mask_words self
  
  function deep_model_ref_AsciiSet (self:t_AsciiSet) : seq t =
    deep_model_AsciiSet self
  
  (* meta rewrite_def function deep_model_ref_AsciiSet *)
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  goal vc_eq_AsciiSet :
    forall self:slice t, other:slice t.
     let other1 = T_AsciiSet'mk other in
     let self1 = T_AsciiSet'mk self in
     inv_AsciiSet self1 /\ inv_AsciiSet other1 ->
     lt (0:t1) (4:t1) &&
     t'int (0:t1) < length (self.elts) &&
     lt (0:t1) (4:t1) &&
     t'int (0:t1) < length (other.elts) &&
     (not get (self.elts) (t'int (0:t1)) = get (other.elts) (t'int (0:t1)) ->
      false <-> mask_words self1 = mask_words other1) /\
     (get (self.elts) (t'int (0:t1)) = get (other.elts) (t'int (0:t1)) ->
      lt (1:t1) (4:t1) &&
      t'int (1:t1) < length (self.elts) &&
      lt (1:t1) (4:t1) &&
      t'int (1:t1) < length (other.elts) &&
      (not get (self.elts) (t'int (1:t1)) = get (other.elts) (t'int (1:t1)) ->
       false <-> mask_words self1 = mask_words other1) /\
      (get (self.elts) (t'int (1:t1)) = get (other.elts) (t'int (1:t1)) ->
       lt (2:t1) (4:t1) &&
       t'int (2:t1) < length (self.elts) &&
       lt (2:t1) (4:t1) &&
       t'int (2:t1) < length (other.elts) &&
       (not get (self.elts) (t'int (2:t1)) = get (other.elts) (t'int (2:t1)) ->
        false <-> mask_words self1 = mask_words other1) /\
       (get (self.elts) (t'int (2:t1)) = get (other.elts) (t'int (2:t1)) ->
        lt (3:t1) (4:t1) &&
        t'int (3:t1) < length (self.elts) &&
        lt (3:t1) (4:t1) &&
        t'int (3:t1) < length (other.elts) &&
        (get (self.elts) (t'int (3:t1)) = get (other.elts) (t'int (3:t1)) <->
         mask_words self1 = mask_words other1))))
  
end
