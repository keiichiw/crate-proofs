theory Coma
  
  (* use why3.BuiltIn.BuiltIn *)
  
  (* use why3.Bool.Bool *)
  
  (* use why3.Unit.Unit *)
  
  (* use creusot.int.UInt8 *)
  
  (* use seq.Seq *)
  
  (* use map.Map *)
  
  function index_Mapping_u8_Int (self:t -> int) (a:t) : int = get self a
  
  (* meta rewrite_def function index_Mapping_u8_Int *)
  
  function map_u8 (seq t) (t -> int) : seq int
  
  axiom map_u8_spec :
    forall self:seq t, m:t -> int [map_u8 self m].
     length (map_u8 self m) = length self
  
  axiom map_u8_spec'0 :
    forall self:seq t, m:t -> int [map_u8 self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get1 (map_u8 self m) i = index_Mapping_u8_Int m (get1 self i)
  
  function byte_sequence_as_ints (bytes:seq t) : seq int =
    map_u8 bytes (fun (byte:t) -> t'int byte)
  
  predicate gt_log_Int (self:int) (other:int) = other < self
  
  (* meta rewrite_def predicate gt_log_Int *)
  
  function tail_u8 (self:seq t) : seq t = self[1..length self]
  
  predicate nonempty_left_concat_tail_u8 (seq t) (seq t)
  
  axiom nonempty_left_concat_tail_u8_spec :
    forall left:seq t, right:seq t [nonempty_left_concat_tail_u8 left right].
     gt_log_Int (length left) 0 -> nonempty_left_concat_tail_u8 left right
  
  axiom nonempty_left_concat_tail_u8_spec'0 :
    forall left:seq t, right:seq t.
     gt_log_Int (length left) 0 ->
     tail_u8 (left ++ right) = (tail_u8 left ++ right)
  
  predicate ge_log_Int (self:int) (other:int) = other <= self
  
  (* meta rewrite_def predicate ge_log_Int *)
  
  predicate well_founded_relation_Int (self:int) (other:int) =
    ge_log_Int self 0 /\ gt_log_Int self other
  
  (* meta rewrite_def predicate well_founded_relation_Int *)
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  constant left : seq t
  
  constant right : seq t
  
  predicate byte_sequence_as_ints_concat (seq t) (seq t)
  
  goal vc_byte_sequence_as_ints_concat :
    if length left = 0
    then true /\
         byte_sequence_as_ints (left ++ right) =
         (byte_sequence_as_ints left ++ byte_sequence_as_ints right)
    else gt_log_Int (length left) 0 /\
         (nonempty_left_concat_tail_u8 left right /\
          tail_u8 (left ++ right) = (tail_u8 left ++ right) ->
          (if nonempty_left_concat_tail_u8 left right
           then well_founded_relation_Int (length left)
                (length (tail_u8 left)) /\
                (byte_sequence_as_ints_concat (tail_u8 left) right /\
                 byte_sequence_as_ints (tail_u8 left ++ right) =
                 (byte_sequence_as_ints (tail_u8 left)
                  ++ byte_sequence_as_ints right) ->
                 byte_sequence_as_ints_concat (tail_u8 left) right /\
                 byte_sequence_as_ints (left ++ right) =
                 (byte_sequence_as_ints left ++ byte_sequence_as_ints right))
           else false /\
                byte_sequence_as_ints (left ++ right) =
                (byte_sequence_as_ints left ++ byte_sequence_as_ints right)))
