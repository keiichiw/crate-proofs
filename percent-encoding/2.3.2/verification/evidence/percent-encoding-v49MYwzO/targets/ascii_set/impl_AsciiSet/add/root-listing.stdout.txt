  goal vc_add :

       t'int result = size_of_u32 ->

        not const_BITS_PER_CHUNK = (0:t1) &&
        (not const_BITS_PER_CHUNK = zeros \/

          not t'int const_BITS_PER_CHUNK = 0) /\

            length (result1.elts) = length (self.elts) ->

            [t'int (udiv (of_BV256 (to_BV256 byte)) const_BITS_PER_CHUNK)] =
            bw_or
            (get (self.elts)
             (t'int (udiv (of_BV256 (to_BV256 byte)) const_BITS_PER_CHUNK)))

                    (shl (1:t) (of_BV2561 (to_BV256 (bw_and byte (31:t2)))))
               else get (mask_words (T_AsciiSet'mk result1)) i =
                    get (mask_words self1) i)))))))
