  function utf8_step (bytes:seq int) (pos:int) : tup3_Int_Int_bool =
    let b0 = get bytes pos in
    let remaining = length bytes - pos in
    if 0 <= b0 /\ b0 <= 127 then Tup3_Int_Int_bool'mk b0 1 True
    else if 194 <= b0 /\ b0 <= 223
         then if remaining < 2
              then Tup3_Int_Int_bool'mk 65533 remaining False
              else let b1 = get bytes (pos + 1) in
                   if 128 <= b1 /\ b1 <= 191
                   then Tup3_Int_Int_bool'mk (((b0 - 192) * 64) + (b1 - 128))
                        2 True
                   else Tup3_Int_Int_bool'mk 65533 1 False
         else if 224 <= b0 /\ b0 <= 239
              then if remaining < 2
                   then Tup3_Int_Int_bool'mk 65533 remaining False
                   else let b1 = get bytes (pos + 1) in
                        let second_ok =
                          (if if b0 = 224 then 160 <= b1 /\ b1 <= 191

  function strict_error (seq int) : t_Option_tup2_Int_Option_Int
  
  axiom strict_error_def :
    forall bytes:seq int [strict_error bytes].
     strict_error bytes =
     (if length bytes = 0 then None'0
      else match utf8_step bytes 0 with
      | Tup3_Int_Int_bool'mk _ consumed valid ->
          if valid = True
          then match strict_error bytes[consumed..length bytes] with
          | Some'0 (Tup2_Int_Option_Int'mk offset len) ->
              Some'0 (Tup2_Int_Option_Int'mk (consumed + offset) len)
          | None'0 -> None'0
          end
          else let lead = get bytes 0 in
               let remaining = length bytes in
               let truncated =
                 (if 194 <= lead /\ lead <= 223 /\ remaining < 2 \/
                     224 <= lead /\
                     lead <= 239 /\ remaining < 3 /\ consumed = remaining \/

                     lead <= 244 /\ remaining < 4 /\ consumed = remaining
                  then True else False)
               in
               Some'0
               (Tup2_Int_Option_Int'mk 0
                (if truncated = True then None else Some consumed))
      end)
