  type t_AsciiSet
  
  type t_PercentEncode =
    | T_PercentEncode'mk (bytes:slice t) (ascii_set:t_AsciiSet)
  
  (* meta model_record function ascii_set *)
  
  (* meta model_record function bytes *)
  
  predicate gt_log_Int (self:int) (other:int) = other < self
  
  (* meta rewrite_def predicate gt_log_Int *)
  
  predicate ge_log_Int (self:int) (other:int) = other <= self
  
  (* meta rewrite_def predicate ge_log_Int *)
  
  predicate should_encode_model (mask:seq t1) (byte:t) =
    ge_log_Int (t'int byte) 128 \/
    not bw_and (get mask (div (t'int byte) 32))
        (shl (1:t1) (of_int (t'int (bw_and1 byte (31:t))))) = (0:t1)
  
  function uppercase_hex (digit:int) : int =
    if digit < 10 then 48 + digit else 55 + digit
  
  axiom uppercase_hex_spec :
    forall digit:int [uppercase_hex digit].
     uppercase_hex digit = (if digit < 10 then 48 + digit else 55 + digit)
  
  function uppercase_hex'0 (digit:int) : int = uppercase_hex digit
  
  function encoded_byte_model (byte:int) : seq int =
    (singleton 37 ++ singleton (uppercase_hex'0 (div byte 16)))
    ++ singleton (uppercase_hex'0 (mod byte 16))
  
  function safe_run_end (seq t) int (seq t1) : int
  
  axiom safe_run_end_spec :
    forall input:seq t, start:int, mask:seq t1
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start <= safe_run_end input start mask /\
     safe_run_end input start mask <= length input
  
  axiom safe_run_end_spec'0 :
    forall input:seq t, start:int, mask:seq t1
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     (forall i:int.
       start <= i /\ i < safe_run_end input start mask ->
       not should_encode_model mask (get input i))
  
  axiom safe_run_end_spec'1 :
    forall input:seq t, start:int, mask:seq t1
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     safe_run_end input start mask = length input \/
     should_encode_model mask (get input (safe_run_end input start mask))
  
  axiom safe_run_end_spec'2 :
    forall input:seq t, start:int, mask:seq t1
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start < length input /\ not should_encode_model mask (get input start) ->
     start < safe_run_end input start mask
  
  function index_Mapping_u8_Int (self:t -> int) (a:t) : int = get1 self a
  
  (* meta rewrite_def function index_Mapping_u8_Int *)
  
  function map_u8 (seq t) (t -> int) : seq int
  
  axiom map_u8_spec :
    forall self:seq t, m:t -> int [map_u8 self m].
     length (map_u8 self m) = length self
  
  axiom map_u8_spec'0 :
    forall self:seq t, m:t -> int [map_u8 self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get (map_u8 self m) i = index_Mapping_u8_Int m (get self i)
  
  function byte_sequence_as_ints (bytes'0:seq t) : seq int =
    map_u8 bytes'0 (fun (byte:t) -> t'int byte)
  
  function encoded_chunks (seq t1) (seq t) : seq (seq int)
  
  axiom encoded_chunks_def :
    forall mask:seq t1, input:seq t [encoded_chunks mask input].
     length mask = 4 ->
     encoded_chunks mask input =
     (if length input = 0 then empty: seq (seq int)
      else if should_encode_model mask (get input 0)
           then singleton (encoded_byte_model (t'int (get input 0)))
                ++ encoded_chunks mask input[1..length input]
           else let end' = safe_run_end input 0 mask in
                singleton (byte_sequence_as_ints input[0..end'])
                ++ encoded_chunks mask input[end'..length input])
  
  axiom encoded_chunks_spec :
    forall mask:seq t1, input:seq t [encoded_chunks mask input].
     length mask = 4 ->
     (if length input = 0 then length (encoded_chunks mask input) = 0
      else gt_log_Int (length (encoded_chunks mask input)) 0 /\
           (if should_encode_model mask (get input 0)
            then get (encoded_chunks mask input) 0 =
                 encoded_byte_model (t'int (get input 0)) /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[1..length input]
            else let end' = safe_run_end input 0 mask in
                 get (encoded_chunks mask input) 0 =
                 byte_sequence_as_ints input[0..end'] /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[end'..length input]))
  
  function mask_words t_AsciiSet : seq t1
  
  axiom mask_words_spec :
    forall self:t_AsciiSet [mask_words self]. length (mask_words self) = 4
  
  function configured_set (self:t_PercentEncode) : seq t1 =
    mask_words (self.ascii_set)
  
  (* meta rewrite_def function configured_set *)
  
  function remaining_input (self:t_PercentEncode) : seq t = view (self.bytes)
  
  (* meta rewrite_def function remaining_input *)
  
  function index_Mapping_ref_str_Seq_Int (self:string -> seq int) (a:
    string) : seq int = get1 self a
  
  (* meta rewrite_def function index_Mapping_ref_str_Seq_Int *)
  
  function map_ref_str (seq string) (string -> seq int) : seq (seq int)
  
  axiom map_ref_str_spec :
    forall self:seq string, m:string -> seq int [map_ref_str self m].
     length (map_ref_str self m) = length self
  
  axiom map_ref_str_spec'0 :
    forall self:seq string, m:string -> seq int [map_ref_str self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get (map_ref_str self m) i =
      index_Mapping_ref_str_Seq_Int m (get self i)
  
  function tail_char (self:seq t2) : seq t2 = self[1..length self]
  
  function flat_map_char (seq t2) (t2 -> seq t) : seq t
  
  axiom flat_map_char_def :
    forall self:seq t2, other:t2 -> seq t [flat_map_char self other].
     flat_map_char self other =
     (if length self = 0 then empty: seq t
      else get1 other (get self 0) ++ flat_map_char (tail_char self) other)
  
  function to_utf8_char t2 : seq t
  
  axiom to_utf8_char_spec :
    forall self:t2 [to_utf8_char self].
     1 <= length (to_utf8_char self) /\ length (to_utf8_char self) <= 4
  
  function to_bytes (self:seq t2) : seq t =
    flat_map_char self (fun (c:t2) -> to_utf8_char c)
  
  function view_str string : seq t2
  
  function string_chunk_bytes (s:string) : seq int =
    map_u8 (to_bytes (view_str s)) (fun (b:t) -> t'int b)
  
  predicate produces_PercentEncode (self:t_PercentEncode) (visited:seq
    string) (o:t_PercentEncode) =
    encoded_chunks (configured_set self) (remaining_input self) =
    (map_ref_str visited (fun (chunk:string) -> string_chunk_bytes chunk)
     ++ encoded_chunks (configured_set o) (remaining_input o))
  
  predicate string_chunk_map_concat (seq string) (seq string)
  
  axiom string_chunk_map_concat_spec :
    forall ab:seq string, bc:seq string [string_chunk_map_concat ab bc].
     string_chunk_map_concat ab bc
  
  axiom string_chunk_map_concat_spec'0 :
    forall ab:seq string, bc:seq string.
     map_ref_str (ab ++ bc) (fun (s:string) -> string_chunk_bytes s) =
     (map_ref_str ab (fun (s:string) -> string_chunk_bytes s)
      ++ map_ref_str bc (fun (s:string) -> string_chunk_bytes s))
  
  function concat_assoc_Seq_Int (seq (seq int)) (seq (seq int)) (seq (seq
    int)) : unit
  
  axiom concat_assoc_Seq_Int_spec :
    forall self:seq (seq int), other1:seq (seq int), other2:seq (seq int).
     ((self ++ other1) ++ other2) = (self ++ (other1 ++ other2))

  function produces_trans_PercentEncode t_PercentEncode (seq string)
    t_PercentEncode (seq string) t_PercentEncode : unit
  
  goal vc_produces_trans_PercentEncode :
    produces_PercentEncode a ab b ->
    produces_PercentEncode b bc c ->
    string_chunk_map_concat ab bc /\
    map_ref_str (ab ++ bc) (fun (s:string) -> string_chunk_bytes s) =
    (map_ref_str ab (fun (s:string) -> string_chunk_bytes s)
     ++ map_ref_str bc (fun (s:string) -> string_chunk_bytes s)) ->
    string_chunk_map_concat ab bc && produces_PercentEncode a (ab ++ bc) c
