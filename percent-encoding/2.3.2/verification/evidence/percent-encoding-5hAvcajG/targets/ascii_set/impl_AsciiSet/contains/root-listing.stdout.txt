  constant size_of_u32 : int = 4

  goal vc_size_of_u32'0 : true

    | Tup2_usize_bool'mk (f0:t) (f1:bool)
  
  (* meta model_record function f1 *)
  
  (* meta model_record function f0 *)
  
  constant const_BITS_PER_CHUNK : t
  
  goal vc_set_BITS_PER_CHUNK : true
  
  type t_AsciiSet =
    | T_AsciiSet'mk (mask:slice t1)
  
  (* meta model_record function mask *)
  
  predicate inv_Seq_u32 (_1:seq t1) = true
  
  (* meta rewrite_def predicate inv_Seq_u32 *)
  
  predicate invariant_array_u32_4 (self:slice t1) =
    inv_Seq_u32 (view self) /\ length (view self) = t'int (4:t)
  
  predicate inv_array_u32_4 (_1:slice t1) = invariant_array_u32_4 _1
  
  (* meta rewrite_def predicate inv_array_u32_4 *)

    create 4 (fun (i:int) -> get (view (self.mask)) i)
  
  axiom mask_words_spec :
    forall self:t_AsciiSet [mask_words self]. length (mask_words self) = 4
  
  predicate contains_model (set:t_AsciiSet) (byte:t2) =
    t'int1 byte < 128 /\
    not bw_and (get (mask_words set) (div (t'int1 byte) 32))
        (shl (1:t1) (of_BV256 (to_BV256 (bw_and1 byte (31:t2))))) = (0:t1)
  
  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  goal vc_contains :
    forall self:slice t1, byte:t2.
     let self1 = T_AsciiSet'mk self in
     inv_AsciiSet self1 /\ t'int1 byte < 128 ->
     (forall result:t.
       t'int result = size_of_u32 ->
       not match mul_with_overflow (8:t) result with
           | _fst, _snd -> _snd
           end = True &&
       (const_BITS_PER_CHUNK =
        match mul_with_overflow (8:t) result with

        end ->
        not const_BITS_PER_CHUNK = (0:t) &&
        (not const_BITS_PER_CHUNK = zeros \/
         not t'int const_BITS_PER_CHUNK = 0) /\
        (t'int (udiv (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK) =

         not const_BITS_PER_CHUNK = (0:t) &&
         (not const_BITS_PER_CHUNK = zeros \/
          not t'int const_BITS_PER_CHUNK = 0) /\
         (t'int (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK) =
          mod (t'int (of_BV2561 (to_BV256 byte)))
          (t'int const_BITS_PER_CHUNK) ->
          (uge
           (to_BV2561 (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK))
           (0:t3) /\
           ult
           (to_BV2561 (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK))
           (of_int size) \/
           t'int2
           (to_BV2561 (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK))
           >= 0 /\
           t'int2
           (to_BV2561 (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK))
           < size) /\
          (not bw_and
               (get (self.elts)
                (t'int
                 (udiv (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK)))
               (shl (1:t1)
                (of_BV256
                 (to_BV2561
                  (urem (of_BV2561 (to_BV256 byte)) const_BITS_PER_CHUNK)))) =
               (0:t1) <->
           contains_model self1 byte)))))
