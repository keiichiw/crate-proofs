  (* use why3.BuiltIn.BuiltIn *)

  (* use creusot.prelude.Any *)
  
  type t_Option_bool =
    | None
    | Some bool
  
  type t_DebugTrace =
    | T_DebugTrace'mk (struct_name:string) (field_names:seq string)
        (field_type_tags:seq t) (word_values:seq (seq t1)) (byte_values:seq
        (seq t)) (iter_values:seq (seq t)) (ascii_set_values:seq (seq t1))
        (finish_status:t_Option_bool)
  
  (* meta model_record function finish_status *)
  
  (* meta model_record function ascii_set_values *)
  
  (* meta model_record function iter_values *)
  
  (* meta model_record function byte_values *)
  
  (* meta model_record function word_values *)
  
  (* meta model_record function field_type_tags *)
  
  (* meta model_record function field_names *)
  
  (* meta model_record function struct_name *)
  
  goal vc_empty : true
  
  goal vc_borrow_mut_DebugTrace : true
  
  type t_Formatter
  
  predicate inv_Formatter t_Formatter
  
  type t_DebugStruct
  
  predicate invariant_refmut_Formatter (self:t2 t_Formatter) =
    inv_Formatter (self.current) /\ inv_Formatter (self.final)
  
  (* meta rewrite_def predicate invariant_refmut_Formatter *)
  
  predicate inv_refmut_Formatter (_1:t2 t_Formatter) =
    invariant_refmut_Formatter _1
  
  (* meta rewrite_def predicate inv_refmut_Formatter *)
  
  predicate inv_DebugStruct t_DebugStruct
  
  goal vc_start : true
  
  type t_AsciiSet
  
  type t_PercentEncode =
    | T_PercentEncode'mk (bytes:slice t) (ascii_set:t_AsciiSet)
  
  (* meta model_record function ascii_set *)
  
  (* meta model_record function bytes *)
  
  predicate invariant_refmut_DebugStruct (self:t2 t_DebugStruct) =
    inv_DebugStruct (self.current) /\ inv_DebugStruct (self.final)
  
  (* meta rewrite_def predicate invariant_refmut_DebugStruct *)
  
  predicate inv_refmut_DebugStruct (_1:t2 t_DebugStruct) =
    invariant_refmut_DebugStruct _1
  
  (* meta rewrite_def predicate inv_refmut_DebugStruct *)
  
  goal vc_field_bytes : true
  
  predicate resolve_refmut_DebugStruct (_1:t2 t_DebugStruct) =
    _1.final = _1.current
  
  (* meta rewrite_def predicate resolve_refmut_DebugStruct *)
  
  predicate inv_AsciiSet t_AsciiSet

  (* meta rewrite_def predicate invariant_ref_AsciiSet *)
  
  predicate inv_ref_AsciiSet (_1:t_AsciiSet) = invariant_ref_AsciiSet _1
  
  (* meta rewrite_def predicate inv_ref_AsciiSet *)
  
  function mask_words t_AsciiSet : seq t1
  
  axiom mask_words_spec :
    forall self:t_AsciiSet [mask_words self]. length (mask_words self) = 4
  
  goal vc_field_ascii_set : true

    | Ok unit
    | Err unit
  
  goal vc_finish : true
  
  predicate resolve_DebugStruct t_DebugStruct
  
  predicate resolve_refmut_Formatter (_1:t2 t_Formatter) =
    _1.final = _1.current
  
  (* meta rewrite_def predicate resolve_refmut_Formatter *)
  
  predicate inv_PercentEncode t_PercentEncode
  
  axiom inv_axiom :
    forall x:t_PercentEncode [inv_PercentEncode x].
     inv_PercentEncode x <-> inv_ref_AsciiSet (x.ascii_set)

    inv_PercentEncode self
  
  (* meta rewrite_def predicate invariant_ref_PercentEncode *)
  
  predicate inv_ref_PercentEncode (_1:t_PercentEncode) =
    invariant_ref_PercentEncode _1

  (* meta compute_max_steps 1000000 *)
  
  (* meta select_lsinst "all" *)
  
  goal vc_fmt_PercentEncode :
    forall self:t_PercentEncode, formatter:t2 t_Formatter.
     inv_ref_PercentEncode self /\ inv_refmut_Formatter formatter ->
     (forall fin:t_DebugTrace.
       (forall fin1:t_Formatter.
         inv_Formatter fin1 ->
         inv_refmut_Formatter
         (t'mk (formatter.current) fin1 (get_id formatter)) /\
         (forall result:t_DebugStruct.
           inv_DebugStruct result /\
           fin.struct_name = ("PercentEncode":string) /\
           length (fin.field_names) = 0 /\
           length (fin.field_type_tags) = 0 /\
           length (fin.word_values) = 0 /\
           length (fin.byte_values) = 0 /\
           length (fin.iter_values) = 0 /\
           length (fin.ascii_set_values) = 0 /\ fin.finish_status = None ->
           (forall fin2:t_DebugStruct, id:int.
             inv_DebugStruct fin2 ->
             (forall fin3:t_DebugTrace.
               (forall fin4:t_DebugStruct.
                 inv_DebugStruct fin4 ->
                 inv_refmut_DebugStruct
                 (t'mk result fin4 (get_id (t'mk result fin2 id))) /\
                 (fin3.field_names = snoc (fin.field_names) ("bytes":string) /\
                  fin3.field_type_tags = snoc (fin.field_type_tags) (1:t) /\
                  fin3.byte_values =
                  snoc (fin.byte_values) (view (self.bytes)) /\
                  fin3.struct_name = fin.struct_name /\
                  fin3.word_values = fin.word_values /\
                  fin3.iter_values = fin.iter_values /\
                  fin3.ascii_set_values = fin.ascii_set_values /\
                  fin3.finish_status = fin.finish_status ->
                  (resolve_refmut_DebugStruct (t'mk fin4 fin2 id) ->
                   (forall fin5:t_DebugStruct, id1:int.
                     inv_DebugStruct fin5 ->
                     (forall fin6:t_DebugTrace.
                       (forall fin7:t_DebugStruct.
                         inv_DebugStruct fin7 ->
                         (inv_refmut_DebugStruct
                          (t'mk fin2 fin7 (get_id (t'mk fin2 fin5 id1))) /\
                          inv_ref_AsciiSet (self.ascii_set)) /\
                         (fin6.field_names =
                          snoc (fin3.field_names) ("ascii_set":string) /\
                          fin6.field_type_tags =
                          snoc (fin3.field_type_tags) (3:t) /\
                          fin6.ascii_set_values =
                          snoc (fin3.ascii_set_values)
                          (mask_words (self.ascii_set)) /\
                          fin6.struct_name = fin3.struct_name /\
                          fin6.word_values = fin3.word_values /\
                          fin6.byte_values = fin3.byte_values /\
                          fin6.iter_values = fin3.iter_values /\
                          fin6.finish_status = fin3.finish_status ->
                          (resolve_refmut_DebugStruct (t'mk fin7 fin5 id1) ->
                           (forall fin8:t_DebugStruct, id2:int.
                             inv_DebugStruct fin8 ->
                             (forall fin9:t_DebugTrace.
                               (forall fin10:t_DebugStruct.
                                 inv_DebugStruct fin10 ->
                                 inv_refmut_DebugStruct
                                 (t'mk fin5 fin10
                                  (get_id (t'mk fin5 fin8 id2))) /\
                                 (forall result1:t_Result_unit_Error.
                                   fin9.finish_status =
                                   Some
                                   match result1 with
                                   | Ok () -> True
                                   | Err _ -> False
                                   end /\
                                   fin9.struct_name = fin6.struct_name /\
                                   fin9.field_names = fin6.field_names /\
                                   fin9.field_type_tags =
                                   fin6.field_type_tags /\
                                   fin9.word_values = fin6.word_values /\
                                   fin9.byte_values = fin6.byte_values /\
                                   fin9.iter_values = fin6.iter_values /\
                                   fin9.ascii_set_values =
                                   fin6.ascii_set_values ->
                                   (resolve_refmut_DebugStruct
                                    (t'mk fin10 fin8 id2) ->
                                    (resolve_DebugStruct fin8 ->
                                     (resolve_refmut_Formatter
                                      (t'mk fin1 (formatter.final)
                                       (formatter.id3)) ->
                                      fin9.struct_name =
                                      ("PercentEncode":string) &&
                                      length (fin9.field_names) = 2 &&
                                      get (fin9.field_names) 0 =
                                      ("bytes":string) &&
                                      get (fin9.field_names) 1 =
                                      ("ascii_set":string) &&
                                      get (fin9.field_type_tags) 0 = (1:t) &&
                                      get (fin9.field_type_tags) 1 = (3:t) &&
                                      length (fin9.byte_values) = 1 &&
                                      get (fin9.byte_values) 0 =
                                      view (self.bytes) &&
                                      length (fin9.word_values) = 0 &&
                                      length (fin9.iter_values) = 0 &&
                                      length (fin9.ascii_set_values) = 1 &&
                                      get (fin9.ascii_set_values) 0 =
                                      mask_words (self.ascii_set) &&
                                      fin9.finish_status =
                                      Some
                                      match result1 with
                                      | Ok () -> True
                                      | Err _ -> False
                                      end) /\
                                     inv_refmut_Formatter
                                     (t'mk fin1 (formatter.final)
                                      (formatter.id3))) /\
                                    inv_DebugStruct fin8) /\
                                   inv_refmut_DebugStruct
                                   (t'mk fin10 fin8 id2))) /\
                               inv_DebugStruct fin5)) /\
                           inv_DebugStruct fin5) /\
                          inv_refmut_DebugStruct (t'mk fin7 fin5 id1))) /\
                       inv_DebugStruct fin2)) /\
                   inv_DebugStruct fin2) /\
                  inv_refmut_DebugStruct (t'mk fin4 fin2 id))) /\
               inv_DebugStruct result)) /\
           inv_DebugStruct result)) /\
       inv_Formatter (formatter.current))
