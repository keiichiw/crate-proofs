  type t_AsciiSet
  
  type t_PercentEncode =
    | T_PercentEncode'mk (bytes:slice t) (ascii_set:t_AsciiSet)
  
  (* meta model_record function ascii_set *)
  
  (* meta model_record function bytes *)
  
  type t_String
  
  type t_Cow_str =
    | Borrowed string
    | Owned t_String
  
  predicate inv_AsciiSet t_AsciiSet
  
  predicate invariant_ref_AsciiSet (self:t_AsciiSet) = inv_AsciiSet self
  
  (* meta rewrite_def predicate invariant_ref_AsciiSet *)
  
  predicate inv_ref_AsciiSet (_1:t_AsciiSet) = invariant_ref_AsciiSet _1
  
  (* meta rewrite_def predicate inv_ref_AsciiSet *)
  
  predicate inv_PercentEncode t_PercentEncode
  
  axiom inv_axiom :
    forall x:t_PercentEncode [inv_PercentEncode x].
     inv_PercentEncode x <-> inv_ref_AsciiSet (x.ascii_set)
  
  predicate inv_String t_String
  
  predicate inv_Cow_str t_Cow_str
  
  axiom inv_axiom'0 :
    forall x:t_Cow_str [inv_Cow_str x].
     inv_Cow_str x <->
     match x with
     | Borrowed f0 -> true
     | Owned f0 -> inv_String f0
     end
  
  function index_Mapping_u8_Int (self:t -> int) (a:t) : int = get self a
  
  (* meta rewrite_def function index_Mapping_u8_Int *)
  
  function map_u8 (seq t) (t -> int) : seq int
  
  axiom map_u8_spec :
    forall self:seq t, m:t -> int [map_u8 self m].
     length (map_u8 self m) = length self
  
  axiom map_u8_spec'0 :
    forall self:seq t, m:t -> int [map_u8 self m].
     forall i:int.
      0 <= i /\ i < length self ->
      get1 (map_u8 self m) i = index_Mapping_u8_Int m (get1 self i)
  
  function byte_sequence_as_ints (bytes'0:seq t) : seq int =
    map_u8 bytes'0 (fun (byte:t) -> t'int byte)
  
  function tail_char (self:seq t1) : seq t1 = self[1..length self]
  
  function flat_map_char (seq t1) (t1 -> seq t) : seq t
  
  axiom flat_map_char_def :
    forall self:seq t1, other:t1 -> seq t [flat_map_char self other].
     flat_map_char self other =
     (if length self = 0 then empty: seq t
      else get other (get1 self 0) ++ flat_map_char (tail_char self) other)
  
  function to_utf8_char t1 : seq t
  
  axiom to_utf8_char_spec :
    forall self:t1 [to_utf8_char self].
     1 <= length (to_utf8_char self) /\ length (to_utf8_char self) <= 4
  
  function to_bytes (self:seq t1) : seq t =
    flat_map_char self (fun (c:t1) -> to_utf8_char c)
  
  function view_str string : seq t1
  
  function view_String t_String : seq t1
  
  function cow_string_bytes (value:t_Cow_str) : seq int =
    match value with
    | Borrowed s -> byte_sequence_as_ints (to_bytes (view_str s))
    | Owned s -> byte_sequence_as_ints (to_bytes (view_String s))
    end
  
  function flatten_chunks (seq (seq int)) : seq int
  
  axiom flatten_chunks_spec :
    forall chunks:seq (seq int) [flatten_chunks chunks].
     if length chunks = 0 then length (flatten_chunks chunks) = 0
     else flatten_chunks chunks =
          (get1 chunks 0 ++ flatten_chunks chunks[1..length chunks])
  
  predicate gt_log_Int (self:int) (other:int) = other < self
  
  (* meta rewrite_def predicate gt_log_Int *)
  
  predicate ge_log_Int (self:int) (other:int) = other <= self
  
  (* meta rewrite_def predicate ge_log_Int *)
  
  predicate should_encode_model (mask:seq t2) (byte:t) =
    ge_log_Int (t'int byte) 128 \/
    not bw_and (get1 mask (div (t'int byte) 32))
        (shl (1:t2) (of_int (t'int (bw_and1 byte (31:t))))) = (0:t2)
  
  function uppercase_hex (digit:int) : int =
    if digit < 10 then 48 + digit else 55 + digit
  
  axiom uppercase_hex_spec :
    forall digit:int [uppercase_hex digit].
     uppercase_hex digit = (if digit < 10 then 48 + digit else 55 + digit)
  
  function uppercase_hex'0 (digit:int) : int = uppercase_hex digit
  
  function encoded_byte_model (byte:int) : seq int =
    (singleton 37 ++ singleton (uppercase_hex'0 (div byte 16)))
    ++ singleton (uppercase_hex'0 (mod byte 16))
  
  function safe_run_end (seq t) int (seq t2) : int
  
  axiom safe_run_end_spec :
    forall input:seq t, start:int, mask:seq t2
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start <= safe_run_end input start mask /\
     safe_run_end input start mask <= length input
  
  axiom safe_run_end_spec'0 :
    forall input:seq t, start:int, mask:seq t2
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     (forall i:int.
       start <= i /\ i < safe_run_end input start mask ->
       not should_encode_model mask (get1 input i))
  
  axiom safe_run_end_spec'1 :
    forall input:seq t, start:int, mask:seq t2
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     safe_run_end input start mask = length input \/
     should_encode_model mask (get1 input (safe_run_end input start mask))
  
  axiom safe_run_end_spec'2 :
    forall input:seq t, start:int, mask:seq t2
     [safe_run_end input start mask].
     0 <= start /\ start <= length input ->
     length mask = 4 ->
     start < length input /\ not should_encode_model mask (get1 input start) ->
     start < safe_run_end input start mask
  
  function encoded_chunks (seq t2) (seq t) : seq (seq int)
  
  axiom encoded_chunks_def :
    forall mask:seq t2, input:seq t [encoded_chunks mask input].
     length mask = 4 ->
     encoded_chunks mask input =
     (if length input = 0 then empty: seq (seq int)
      else if should_encode_model mask (get1 input 0)
           then singleton (encoded_byte_model (t'int (get1 input 0)))
                ++ encoded_chunks mask input[1..length input]
           else let end' = safe_run_end input 0 mask in
                singleton (byte_sequence_as_ints input[0..end'])
                ++ encoded_chunks mask input[end'..length input])
  
  axiom encoded_chunks_spec :
    forall mask:seq t2, input:seq t [encoded_chunks mask input].
     length mask = 4 ->
     (if length input = 0 then length (encoded_chunks mask input) = 0
      else gt_log_Int (length (encoded_chunks mask input)) 0 /\
           (if should_encode_model mask (get1 input 0)
            then get1 (encoded_chunks mask input) 0 =
                 encoded_byte_model (t'int (get1 input 0)) /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[1..length input]
            else let end' = safe_run_end input 0 mask in
                 get1 (encoded_chunks mask input) 0 =
                 byte_sequence_as_ints input[0..end'] /\
                 (encoded_chunks mask input)[1..length
                                                (encoded_chunks mask input)] =
                 encoded_chunks mask input[end'..length input]))
  
  function mask_words t_AsciiSet : seq t2
  
  axiom mask_words_spec :
    forall self:t_AsciiSet [mask_words self]. length (mask_words self) = 4
  
  function configured_set (self:t_PercentEncode) : seq t2 =
    mask_words (self.ascii_set)
  
  (* meta rewrite_def function configured_set *)
  
  function remaining_input (self:t_PercentEncode) : seq t = view (self.bytes)
  
  (* meta rewrite_def function remaining_input *)
  
  goal vc_into_cow : true

  goal vc_from_Cow_str :
    forall iter:t_PercentEncode.
     inv_PercentEncode iter ->
     inv_PercentEncode iter /\
     (forall result:t_Cow_str.
       inv_Cow_str result /\
       cow_string_bytes result =
       flatten_chunks
       (encoded_chunks (configured_set iter) (remaining_input iter)) /\
       match result with
       | Borrowed _ ->
           length
           (encoded_chunks (configured_set iter) (remaining_input iter)) <= 1
       | Owned _ ->
           ge_log_Int
           (length
            (encoded_chunks (configured_set iter) (remaining_input iter)))
           2
       end ->
       inv_Cow_str result /\
       cow_string_bytes result =
       flatten_chunks
       (encoded_chunks (configured_set iter) (remaining_input iter)) /\
       match result with
       | Borrowed _ ->
           length
           (encoded_chunks (configured_set iter) (remaining_input iter)) <= 1
       | Owned _ ->
           ge_log_Int
           (length
            (encoded_chunks (configured_set iter) (remaining_input iter)))
           2
       end)
