                     inv_DebugStruct fin5 ->
                     (forall fin6:t_DebugTrace.
                       (forall fin7:t_DebugStruct.
                         inv_DebugStruct fin7 ->

                         (forall result1:t_Result_unit_Error.
                           fin6.finish_status =
                           Some
                           match result1 with
                           | Ok () -> True
                           | Err _ -> False
                           end /\
                           fin6.struct_name = fin3.struct_name /\
                           fin6.field_names = fin3.field_names /\
                           fin6.field_type_tags = fin3.field_type_tags /\
                           fin6.word_values = fin3.word_values /\
                           fin6.byte_values = fin3.byte_values /\

                             (resolve_refmut_Formatter
                              (t'mk fin1 (formatter.final) (formatter.id2)) ->
                              fin6.struct_name = ("PercentDecode":string) &&
                              length (fin6.field_names) = 1 &&
                              get (fin6.field_names) 0 = ("bytes":string) &&
                              get (fin6.field_type_tags) 0 = (2:t) &&
                              length (fin6.iter_values) = 1 &&

                              view (view_Iter_u8 self) &&
                              length (fin6.word_values) = 0 &&
                              length (fin6.byte_values) = 0 &&
                              length (fin6.ascii_set_values) = 0 &&
                              fin6.finish_status =
                              Some

                              | Ok () -> True
                              | Err _ -> False
                              end) /\

                             (t'mk fin1 (formatter.final) (formatter.id2))) /\

                       inv_DebugStruct fin2)) /\
                   inv_DebugStruct fin2) /\
                  inv_refmut_DebugStruct (t'mk fin4 fin2 id))) /\
               inv_DebugStruct result)) /\
           inv_DebugStruct result)) /\
       inv_Formatter (formatter.current))
